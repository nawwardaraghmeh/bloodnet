    create table BloodPack (
        plasma boolean not null,
        platelets boolean not null,
        redCells boolean not null,
        donation_id bigint not null,
        expiresAt timestamp(6),
        id bigint generated by default as identity,
        producedAt timestamp(6),
        storageLocation_id bigint,
        packCode varchar(255) unique,
        state enum ('AVAILABLE','DISCARDED','ISSUED','QUARANTINED','RESERVED'),
        primary key (id)
    )

    create table Donation (
        volume float(53) not null,
        collectedAt timestamp(6),
        donor_id bigint not null,
        id bigint generated by default as identity,
        collectionSite varchar(255),
        screeningStatus enum ('FAILED','PASSED','PENDING'),
        primary key (id)
    )

    create table Donor (

        fiscalCode varchar(255) unique,
        name varchar(255),
        bloodType enum ('AB_NEGATIVE','AB_POSITIVE','A_NEGATIVE','A_POSITIVE','B_NEGATIVE','B_POSITIVE','O_NEGATIVE','O_POSITIVE'),
        primary key (id)
    )

    create table Issue (
        emergencyRelease boolean not null,
        bloodPack_id bigint not null,
        id bigint generated by default as identity,
        issuedAt timestamp(6),
        patient_id bigint not null,
        ward varchar(255),
        outcome enum ('RETURNED','TRANSFUSED','WASTED'),
        primary key (id)
    )
